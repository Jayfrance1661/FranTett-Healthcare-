/* =====================================================
   SET DATE INPUT
===================================================== */

function setInputDate(
    id,
    value
) {

    const element =
        document.getElementById(
            id
        );


    if (!element) {

        return;

    }


    if (!value) {

        element.value =
            "";

        return;

    }


    element.value =
        String(value).substring(
            0,
            10
        );

}
